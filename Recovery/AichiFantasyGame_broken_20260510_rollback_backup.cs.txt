using System;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.EventSystems;

namespace AichiFantasy
{
    public sealed class AichiFantasyGame : MonoBehaviour
    {
        const string SaveKey = "AichiFantasyTRPG.Progress";
        enum Mode { Title, CharacterSelect, Dice, Scene, Battle, Ending, Unlocks }

        [Serializable] sealed class Progress
        {
            public int memoryFragments;
            public int insuranceTickets;
            public int maxInstabilityUnlocked = 1;
            public List<string> unlockedCharacters = new List<string> { "traveler", "worker", "local", "occult" };
            public List<string> endings = new List<string>();
            public List<string> deaths = new List<string>();
            public List<string> seenMonsters = new List<string>();
            public List<string> monsterWeaknesses = new List<string>();
            public List<string> bossesDefeated = new List<string>();
            public List<string> brokenGear = new List<string>();
            public List<string> regretLog = new List<string>();
            public List<string> milestoneClaims = new List<string>();
            public List<string> warehouseGear = new List<string>();
        }

        sealed class CharacterDef { public string id, name, subtitle, description, weapon, armor, ability; public int unlockCost; public Stats stats; public float rewardRate = 1f; }
        sealed class Stats
        {
            public int maxHp, hp, maxMp, mp, attack, defense, speed, luck, maxSanity, sanity, hunger, money, localKnowledge, misoResistance, machineAptitude, mythosKnowledge, mythosCorruption;
            public Stats Clone() { return (Stats)MemberwiseClone(); }
        }
        sealed class Gear { public string id, name, kind, slot, rarity, setTag, note; public int attack, defense, speed, luck, score; public Gear Clone() { return (Gear)MemberwiseClone(); } }
        sealed class Choice { public string label, next, battle, ending, disabledReason; public Action<RunState> effect; public Func<RunState, bool> condition; }
        sealed class ChoiceCommand { public string label; public Action action; public bool interactable; }
        sealed class SceneDef { public string id, title, area, image, portrait, text; public List<Choice> choices = new List<Choice>(); }
        sealed class EnemyDef { public string id, name, image, intro, victoryText, defeatEnding, weakness, portrait; public int maxHp, hp, attack, defense, speed, sanityDamage, reward; }
        sealed class RunState
        {
            public CharacterDef character; public Stats stats; public Gear weapon, armor, accessory, pendingGear; public string sceneId, battleReturnScene, lastChoiceLabel, lastRollSummary; public int steps, instability = 1, shachiGaze;
            public HashSet<string> flags = new HashSet<string>(); public List<string> recentLog = new List<string>();
        }

        readonly Dictionary<string, CharacterDef> characters = new Dictionary<string, CharacterDef>();
        readonly Dictionary<string, Gear> gears = new Dictionary<string, Gear>();
        readonly Dictionary<string, SceneDef> scenes = new Dictionary<string, SceneDef>();
        readonly Dictionary<string, EnemyDef> enemies = new Dictionary<string, EnemyDef>();
        readonly Dictionary<string, Sprite> spriteCache = new Dictionary<string, Sprite>();
        readonly Dictionary<string, Sprite> portraitCache = new Dictionary<string, Sprite>();
        readonly List<ChoiceCommand> choiceCommands = new List<ChoiceCommand>();
        readonly System.Random rng = new System.Random();

        Progress progress; RunState run; EnemyDef activeEnemy; Mode mode; int lastReward;
        Canvas canvas; Image backgroundImage; Image portraitImage; Text hpText, coinText, titleText, areaText, bodyText, statsText, inventoryText, footerText, battleText; RectTransform choiceRoot, battleRoot; Button[] choiceButtons; Text[] choiceButtonLabels; Slider hpSlider, enemyHpSlider; Button attackButton, guardButton; AudioSource audioSource; AudioClip clickSfx, hitSfx, hurtSfx, doomSfx, rewardSfx;

        void Awake() { BuildData(); LoadProgress(); BuildUi(); LoadAudio(); ShowTitle(); }

        void BuildUi()
        {
            if (FindObjectOfType<EventSystem>() == null) { var es = new GameObject("EventSystem"); es.AddComponent<EventSystem>(); es.AddComponent<StandaloneInputModule>(); }
            audioSource = gameObject.AddComponent<AudioSource>();
            canvas = new GameObject("Canvas").AddComponent<Canvas>(); canvas.renderMode = RenderMode.ScreenSpaceOverlay;
            var scaler = canvas.gameObject.AddComponent<CanvasScaler>(); scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize; scaler.referenceResolution = new Vector2(1280, 720); canvas.gameObject.AddComponent<GraphicRaycaster>();
            backgroundImage = NewImage("Background", canvas.transform, 0, 0, 1, 1); backgroundImage.color = new Color(0.02f, 0.02f, 0.035f);
            portraitImage = NewImage("Portrait", canvas.transform, 0.52f, 0.10f, 1.04f, 0.98f); portraitImage.preserveAspect = true; portraitImage.raycastTarget = false; portraitImage.color = new Color(1, 1, 1, 0.92f);
            var top = NewPanel("TopBar", canvas.transform, 0.02f, 0.91f, 0.42f, 0.985f, new Color(0, 0, 0, 0.58f));
            hpSlider = NewSlider("HpSlider", top, 0.02f, 0.12f, 0.38f, 0.8f); hpText = NewText("HPText", top, "HP", 18, TextAnchor.MiddleLeft, 0.40f, 0.05f, 0.62f, 0.90f); coinText = NewText("CoinText", top, "所持金", 18, TextAnchor.MiddleLeft, 0.64f, 0.05f, 0.98f, 0.90f);
            titleText = NewText("Title", canvas.transform, "", 32, TextAnchor.MiddleLeft, 0.035f, 0.79f, 0.55f, 0.89f); areaText = NewText("Area", canvas.transform, "", 18, TextAnchor.MiddleLeft, 0.04f, 0.735f, 0.58f, 0.79f);
            bodyText = NewText("Body", canvas.transform, "", 21, TextAnchor.UpperLeft, 0.04f, 0.26f, 0.60f, 0.72f); statsText = NewText("Stats", canvas.transform, "", 17, TextAnchor.UpperLeft, 0.68f, 0.55f, 0.98f, 0.89f); inventoryText = NewText("Inventory", canvas.transform, "", 16, TextAnchor.UpperLeft, 0.68f, 0.17f, 0.98f, 0.53f); footerText = NewText("Footer", canvas.transform, "", 15, TextAnchor.MiddleLeft, 0.04f, 0.17f, 0.62f, 0.235f);
            choiceRoot = NewPanel("Choices", canvas.transform, 0.035f, 0.02f, 0.97f, 0.155f, new Color(0, 0, 0, 0.42f)); choiceButtons = new Button[6]; choiceButtonLabels = new Text[6];
            for (int i = 0; i < 6; i++) { var b = NewButton("Choice" + i, choiceRoot, "Choice", new Color(0.12f, 0.13f, 0.16f, 0.94f), 17); int col = i % 3, row = i / 3; SetRect(b.GetComponent<RectTransform>(), 0.01f + col * 0.33f, 0.52f - row * 0.50f, 0.32f + col * 0.33f, 0.96f - row * 0.50f); choiceButtons[i] = b; choiceButtonLabels[i] = b.GetComponentInChildren<Text>(); }
            battleRoot = NewPanel("戦闘", canvas.transform, 0.04f, 0.02f, 0.62f, 0.25f, new Color(0.02f, 0.01f, 0.015f, 0.75f)); battleText = NewText("BattleText", battleRoot, "", 19, TextAnchor.UpperLeft, 0.02f, 0.44f, 0.98f, 0.96f); enemyHpSlider = NewSlider("EnemyHp", battleRoot, 0.02f, 0.32f, 0.98f, 0.42f);
            attackButton = NewButton("AttackButton", battleRoot, "攻撃", new Color(0.45f, 0.09f, 0.09f), 20); SetRect(attackButton.GetComponent<RectTransform>(), 0.02f, 0.06f, 0.48f, 0.28f); guardButton = NewButton("GuardButton", battleRoot, "防御", new Color(0.11f, 0.19f, 0.32f), 20); SetRect(guardButton.GetComponent<RectTransform>(), 0.52f, 0.06f, 0.98f, 0.28f); attackButton.onClick.AddListener(PlayerAttack); guardButton.onClick.AddListener(PlayerGuard); battleRoot.gameObject.SetActive(false);
        }

        void SetRect(RectTransform r, float minX, float minY, float maxX, float maxY) { r.anchorMin = new Vector2(minX, minY); r.anchorMax = new Vector2(maxX, maxY); r.offsetMin = Vector2.zero; r.offsetMax = Vector2.zero; }
        Image NewImage(string name, Transform parent, float minX, float minY, float maxX, float maxY) { var go = new GameObject(name); go.transform.SetParent(parent, false); var r = go.AddComponent<RectTransform>(); SetRect(r, minX, minY, maxX, maxY); return go.AddComponent<Image>(); }
        RectTransform NewPanel(string name, Transform parent, float minX, float minY, float maxX, float maxY, Color color) { var image = NewImage(name, parent, minX, minY, maxX, maxY); image.color = color; return image.rectTransform; }
        Text NewText(string name, Transform parent, string text, int size, TextAnchor anchor, float minX, float minY, float maxX, float maxY) { var go = new GameObject(name); go.transform.SetParent(parent, false); var r = go.AddComponent<RectTransform>(); SetRect(r, minX, minY, maxX, maxY); var t = go.AddComponent<Text>(); t.font = Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf"); t.text = text; t.fontSize = size; t.alignment = anchor; t.color = new Color(0.94f, 0.91f, 0.84f); t.horizontalOverflow = HorizontalWrapMode.Wrap; t.verticalOverflow = VerticalWrapMode.Truncate; return t; }
        Button NewButton(string name, Transform parent, string text, Color color, int size) { var go = new GameObject(name); go.transform.SetParent(parent, false); var r = go.AddComponent<RectTransform>(); SetRect(r, 0, 0, 1, 1); var image = go.AddComponent<Image>(); image.color = color; var b = go.AddComponent<Button>(); var label = NewText("Text", go.transform, text, size, TextAnchor.MiddleCenter, 0.03f, 0.08f, 0.97f, 0.92f); label.color = Color.white; return b; }
        Slider NewSlider(string name, Transform parent, float minX, float minY, float maxX, float maxY) { var go = new GameObject(name); go.transform.SetParent(parent, false); var r = go.AddComponent<RectTransform>(); SetRect(r, minX, minY, maxX, maxY); var s = go.AddComponent<Slider>(); s.interactable = false; var fill = NewImage("Fill", go.transform, 0, 0, 1, 1); fill.color = new Color(0.35f, 0.08f, 0.08f, 0.9f); s.fillRect = fill.rectTransform; s.targetGraphic = fill; return s; }

        void LoadAudio() { clickSfx = Resources.Load<AudioClip>("AichiFantasy/Sfx/click"); hitSfx = Resources.Load<AudioClip>("AichiFantasy/Sfx/hit"); hurtSfx = Resources.Load<AudioClip>("AichiFantasy/Sfx/hurt"); doomSfx = Resources.Load<AudioClip>("AichiFantasy/Sfx/doom"); rewardSfx = Resources.Load<AudioClip>("AichiFantasy/Sfx/reward"); }
        void Play(AudioClip clip, float volume = 0.7f) { if (audioSource != null && clip != null) audioSource.PlayOneShot(clip, volume); }
        void LoadProgress() { progress = PlayerPrefs.HasKey(SaveKey) ? JsonUtility.FromJson<Progress>(PlayerPrefs.GetString(SaveKey)) : new Progress(); if (progress == null) progress = new Progress(); if (progress.unlockedCharacters == null || progress.unlockedCharacters.Count == 0) progress.unlockedCharacters = new List<string> { "traveler", "worker", "local", "occult" }; if (progress.maxInstabilityUnlocked <= 0) progress.maxInstabilityUnlocked = 1; }
        void SaveProgress() { PlayerPrefs.SetString(SaveKey, JsonUtility.ToJson(progress)); PlayerPrefs.Save(); }

        void ShowTitle() { mode = Mode.Title; run = null; SetBackground("title"); SetPortrait("shachi_avatar"); titleText.text = "愛知幻想TRPG"; areaText.text = "搭乗ゲートへ続く周回路"; bodyText.text = "愛知は5つのステージに折りたたまれている。各ステージにはイベント、休憩所、ボスがあり、LUK判定は出目・目標値・成功/失敗・差分まで表示される。"; statsText.text = "記憶片 " + progress.memoryFragments + "\nInsurance " + progress.insuranceTickets + "\nMax risk " + progress.maxInstabilityUnlocked; inventoryText.text = "周回中の最近の出来事がここに表示される。"; footerText.text = "コマンド表示を毎画面作り直す。"; battleRoot.gameObject.SetActive(false); choiceRoot.gameObject.SetActive(true); ClearChoices(); AddChoiceButton("開始", ShowCharacterSelect); AddChoiceButton("記録", ShowUnlocks); AddChoiceButton("初期化", ResetProgress); }
        void ShowCharacterSelect() { mode = Mode.CharacterSelect; SetBackground("characters"); SetPortrait("traveler"); titleText.text = "キャラクター選択"; areaText.text = "探索者を選ぶ"; bodyText.text = "未解放キャラは必要な記憶片を表示する。選ぶとステータスと初期装備を確認できる。"; statsText.text = ""; inventoryText.text = "記憶片 " + progress.memoryFragments; battleRoot.gameObject.SetActive(false); choiceRoot.gameObject.SetActive(true); ClearChoices(); foreach (var pair in characters) { var c = pair.Value; bool unlocked = progress.unlockedCharacters.Contains(c.id) || progress.memoryFragments >= c.unlockCost; AddChoiceButton(unlocked ? c.name : c.name + "\nNeed memory " + c.unlockCost, () => ShowCharacterConfirm(c), unlocked); } AddChoiceButton("Title", ShowTitle); }
        void ShowCharacterConfirm(CharacterDef c) { SetPortrait(c.id); titleText.text = c.name; areaText.text = c.subtitle; bodyText.text = c.description + "\n\nAbility: " + c.ability; statsText.text = FormatStats(c.stats); inventoryText.text = "初期武器: " + GearName(c.weapon) + "\nStart armor: " + GearName(c.armor) + "\nAccessory: none"; ClearChoices(); AddChoiceButton("周回開始", () => StartRun(c)); AddChoiceButton("戻る", ShowCharacterSelect); }
        void StartRun(CharacterDef c) { run = new RunState { character = c, stats = c.stats.Clone(), weapon = GearOrEmpty(c.weapon, "weapon"), armor = GearOrEmpty(c.armor, "armor"), accessory = EmptyGear("accessory"), instability = Mathf.Clamp(progress.maxInstabilityUnlocked, 1, 5) }; run.stats.hp = run.stats.maxHp; run.stats.mp = run.stats.maxMp; LogRun("Run started with " + c.name + ". HP and money displays read from the same run stats."); ShowRunContract(); }
        void ShowRunContract() { SetBackground("dice"); SetPortrait(run.character.id); titleText.text = "開始装備"; areaText.text = "倉庫確認"; bodyText.text = "出発前に持ち込む装備を確認する。空欄は性能0の装備なしとして扱う。"; UpdateSideText(); ClearChoices(); AddChoiceButton("出発", ShowDice); AddChoiceButton("倉庫", ShowWarehouse); AddChoiceButton("キャラクター選択", ShowCharacterSelect); }
        void ShowWarehouse() { titleText.text = "倉庫"; areaText.text = "保管装備"; bodyText.text = progress.warehouseGear.Count == 0 ? "保管装備なし" : string.Join("\n", progress.warehouseGear.ToArray()); ClearChoices(); AddChoiceButton("戻る", ShowRunContract); }
        void ShowDice() { mode = Mode.Dice; SetBackground("dice"); int roll = rng.Next(1, 7) + rng.Next(1, 7); if (roll >= 9) run.stats.luck += 1; LogRun("開始ダイス: " + roll + (roll >= 9 ? " / LUK+1" : " / ボーナスなし")); titleText.text = "開始ダイス"; areaText.text = "さいころが止まる"; bodyText.text = "ダイス結果: " + roll + "\n" + (roll >= 9 ? "よい兆し。LUK+1。" : "鈍い音。ボーナスなし。") + "\n\n今回起きたこと\n" + RecentLog(); UpdateSideText(); ClearChoices(); AddChoiceButton("進む", () => ShowScene("nagoya_after_battle", false)); }
        void ShowScene(string sceneId, bool allowRandom) { mode = Mode.Scene; if (!scenes.ContainsKey(sceneId)) sceneId = "nagoya_after_battle"; var scene = scenes[sceneId]; run.sceneId = sceneId; run.steps++; battleRoot.gameObject.SetActive(false); choiceRoot.gameObject.SetActive(true); SetBackground(scene.image); SetPortrait(scene.portrait); titleText.text = scene.title; areaText.text = scene.area; bodyText.text = scene.text + "\n\nRecent events\n" + RecentLog(); footerText.text = SceneHasLuckChoice(scene) ? "LUK判定は選択後に出目、目標値、成功/失敗、差分を表示する。" : "次の行動を選ぶ。"; UpdateSideText(); ClearChoices(); foreach (var choice in scene.choices) { bool enabled = choice.condition == null || choice.condition(run); AddChoiceButton(choice.label, () => ResolveChoice(choice), enabled); } }
        void ResolveChoice(Choice choice) { run.lastChoiceLabel = choice.label.Replace("\n", " / "); var before = run.stats.Clone(); run.lastRollSummary = ""; choice.effect?.Invoke(run); string delta = BuildStatDelta(before, run.stats); string outcome = run.lastChoiceLabel + "\n" + (string.IsNullOrEmpty(delta) ? "能力変化なし" : delta); if (!string.IsNullOrEmpty(run.lastRollSummary)) outcome += "\n" + run.lastRollSummary; LogRun(outcome.Replace("\n", " / ")); if (!string.IsNullOrEmpty(choice.battle)) StartBattle(choice.battle); else if (!string.IsNullOrEmpty(choice.ending)) ShowEnding(choice.ending); else if (choice.effect != null) ShowChoiceOutcome(choice.next, outcome); else ShowScene(choice.next, false); }
        void ShowChoiceOutcome(string nextSceneId, string outcome) { mode = Mode.Scene; battleRoot.gameObject.SetActive(false); choiceRoot.gameObject.SetActive(true); titleText.text = "行動結果"; areaText.text = "今回起きたこと"; bodyText.text = OutcomeFlavor(outcome) + "\n\n今回起きたこと\n" + outcome; footerText.text = "結果を確認してから進む。"; UpdateSideText(); ClearChoices(); AddChoiceButton("次へ", () => ShowScene(nextSceneId, false)); }
        string OutcomeFlavor(string outcome) { if (outcome.Contains("FAIL")) return "冷たい風が返事をする。それでも選択の痕跡は残った。"; if (outcome.Contains("SUCCESS")) return "一瞬だけ、愛知がこちら側へ傾いた。"; if (outcome.Contains("HP-") || outcome.Contains("SAN-")) return "代償は息と記憶で支払われた。"; return "小さな選択が、次の景色の色を変える。"; }
        int RollLuckDiceAgainst(int target) { int a = rng.Next(1, 7), b = rng.Next(1, 7), c = rng.Next(1, 7), bonus = Mathf.Clamp(run.stats.luck / 2, 0, 6); if (rng.NextDouble() < Mathf.Clamp01(run.stats.luck * 0.055f)) c = 6; int total = a + b + c + bonus, margin = total - target; run.lastRollSummary = "LUK判定: 出目 " + a + "/" + b + "/" + c + " + LUK" + bonus + " = " + total + " / 目標値 " + target + " / " + (margin >= 0 ? "成功 差分+" + margin : "失敗 不足" + (-margin)); return total; }
        bool SceneHasLuckChoice(SceneDef scene) { foreach (var choice in scene.choices) if (!string.IsNullOrEmpty(choice.label) && choice.label.Contains("LUK")) return true; return false; }

        void StartBattle(string enemyId) { if (run != null && run.stats.sanity <= 0) enemyId = "impossible_one"; if (!enemies.ContainsKey(enemyId)) enemyId = "stage_enemy_1"; mode = Mode.Battle; var e = enemies[enemyId]; activeEnemy = new EnemyDef { id = e.id, name = e.name, image = e.image, portrait = e.portrait, maxHp = e.maxHp, hp = e.maxHp, attack = e.attack, defense = e.defense, speed = e.speed, sanityDamage = e.sanityDamage, reward = e.reward, intro = e.intro, victoryText = e.victoryText, defeatEnding = e.defeatEnding, weakness = e.weakness }; if (!progress.seenMonsters.Contains(enemyId)) progress.seenMonsters.Add(enemyId); battleRoot.gameObject.SetActive(true); choiceRoot.gameObject.SetActive(false); SetBackground(e.image); SetPortrait(e.portrait); titleText.text = e.name; areaText.text = "戦闘"; bodyText.text = e.intro + "\n\nWeakness hint: " + e.weakness; UpdateBattleText("戦闘開始。"); UpdateSideText(); SaveProgress(); }
        void PlayerAttack() { if (activeEnemy == null || run == null) return; int damage = Math.Max(1, run.stats.attack + run.weapon.attack + run.accessory.attack + rng.Next(1, 7) - activeEnemy.defense); activeEnemy.hp -= damage; Play(hitSfx); if (activeEnemy.hp <= 0) { WinBattle(); return; } EnemyCounter("Attack " + damage); }
        void PlayerGuard() { if (activeEnemy == null || run == null) return; run.stats.sanity = Math.Min(run.stats.maxSanity, run.stats.sanity + 1); EnemyCounter("防御。SAN+1"); }
        void EnemyCounter(string prefix) { int defense = run.stats.defense + run.armor.defense + run.accessory.defense; int damage = Math.Max(1, activeEnemy.attack + rng.Next(1, 5) - defense); if (activeEnemy.id == "impossible_one") damage = 999; run.stats.hp = Math.Max(0, run.stats.hp - damage); run.stats.sanity = Math.Max(0, run.stats.sanity - activeEnemy.sanityDamage); Play(hurtSfx); UpdateBattleText(prefix + " / Enemy damage HP-" + damage + " SAN-" + activeEnemy.sanityDamage); UpdateSideText(); if (run.stats.hp <= 0) ShowEnding(activeEnemy.defeatEnding); else if (run.stats.sanity <= 0 && activeEnemy.id != "impossible_one") StartBattle("impossible_one"); }
        void WinBattle() { if (activeEnemy == null) return; string defeated = activeEnemy.id; int reward = activeEnemy.reward; run.stats.money += reward; if (!progress.bossesDefeated.Contains(defeated) && defeated.StartsWith("stage_boss_")) progress.bossesDefeated.Add(defeated); if (defeated == "impossible_one") { ShowEnding("impossible_true"); return; } run.pendingGear = RollDrop(defeated, reward); LogRun("撃破: " + activeEnemy.name + ": 所持金+" + reward + ", ドロップ " + GearShortName(run.pendingGear)); activeEnemy = null; SaveProgress(); ShowGearOffer(run.battleReturnScene ?? run.sceneId ?? "nagoya_after_battle"); }
        void UpdateBattleText(string line) { if (activeEnemy == null) return; battleText.text = activeEnemy.name + " HP " + Math.Max(0, activeEnemy.hp) + "/" + activeEnemy.maxHp + "\n" + line; enemyHpSlider.maxValue = activeEnemy.maxHp; enemyHpSlider.value = Mathf.Clamp(activeEnemy.hp, 0, activeEnemy.maxHp); }

        Gear RollDrop(string enemyId, int power) { string[] ids = { "umbrella", "pcbag", "ticket", "wood_sword", "wrench", "jacket", "suit", "raincoat", "morning_badge", "black_ticket", "sealed_coin" }; Gear gear = gears[ids[Math.Abs(enemyId.GetHashCode()) % ids.Length]].Clone(); ApplyDropRarity(gear, SelectDropRarityTier(power), power); gear.id = gear.id + "_drop_" + rng.Next(10000, 99999); return gear; }
        int SelectDropRarityTier(int power) { double roll = rng.NextDouble(); if (roll < 0.04) return 3; if (roll < 0.15) return 2; if (roll < 0.42) return 1; return 0; }
        void ApplyDropRarity(Gear gear, int tier, int power) { string baseName = StripRarityPrefix(gear.name); float m = tier == 3 ? 3.1f : tier == 2 ? 2.15f : tier == 1 ? 1.45f : 1f; gear.attack = Mathf.RoundToInt(gear.attack * m) + (gear.slot == "weapon" ? tier * 2 : 0); gear.defense = Mathf.RoundToInt(gear.defense * m) + (gear.slot == "armor" ? tier * 2 : 0); gear.speed = Mathf.RoundToInt(gear.speed * (1f + tier * 0.25f)); gear.luck = Mathf.RoundToInt(gear.luck * (1f + tier * 0.35f)) + (gear.slot == "accessory" ? tier : 0); gear.score = gear.attack * 3 + gear.defense * 3 + gear.speed * 2 + gear.luck * 2; gear.rarity = tier == 3 ? "Orange" : tier == 2 ? "Purple" : tier == 1 ? "Green" : "White"; gear.name = gear.rarity + " " + baseName; gear.note += " Transfer success " + TransferChanceText(gear) + "."; }
        string StripRarityPrefix(string name) { foreach (var prefix in new[] { "Orange ", "Purple ", "Green ", "White " }) if (name.StartsWith(prefix)) return name.Substring(prefix.Length); return name; }
        void ShowGearOffer(string next) { if (run.pendingGear == null) { ShowScene(next, false); return; } var current = CurrentGearForSlot(run.pendingGear.slot); battleRoot.gameObject.SetActive(false); choiceRoot.gameObject.SetActive(true); titleText.text = "装備ドロップ"; areaText.text = run.pendingGear.rarity + " / " + run.pendingGear.slot; bodyText.text = GearComparison(run.pendingGear, current); UpdateSideText(); ClearChoices(); AddChoiceButton("装備", () => { EquipGear(run.pendingGear); LogRun("Equipped " + GearShortName(run.pendingGear)); run.pendingGear = null; UpdateSideText(); ShowScene(next, false); }); AddChoiceButton("Warehouse\n" + TransferChanceText(run.pendingGear), () => TryStorePendingGear(next)); AddChoiceButton("捨てる", () => { run.pendingGear = null; ShowScene(next, false); }); }
        Gear CurrentGearForSlot(string slot) { if (slot == "armor") return run.armor; if (slot == "accessory") return run.accessory; return run.weapon; }
        void EquipGear(Gear gear) { if (gear.slot == "armor") run.armor = gear; else if (gear.slot == "accessory") run.accessory = gear; else run.weapon = gear; }
        void TryStorePendingGear(string next) { var gear = run.pendingGear; run.pendingGear = null; float chance = WarehouseTransferChance(gear); bool success = rng.NextDouble() < chance; if (success) { progress.warehouseGear.Add(SerializeGear(gear)); LogRun("倉庫転送成功: " + GearShortName(gear)); bodyText.text = gear.name + "を倉庫へ送った。\n成功率 " + Mathf.RoundToInt(chance * 100f) + "%"; } else { progress.brokenGear.Add(GearShortName(gear) + " 転送失敗"); LogRun("倉庫転送失敗: " + GearShortName(gear)); bodyText.text = gear.name + "は転送中に破損した。\n成功率 " + Mathf.RoundToInt(chance * 100f) + "%"; } SaveProgress(); ClearChoices(); AddChoiceButton("次へ", () => ShowScene(next, false)); }

        string TransferChanceText(Gear gear) { return Mathf.RoundToInt(WarehouseTransferChance(gear) * 100f) + "%"; }

        string BuildStatDelta(Stats before, Stats after) { var parts = new List<string>(); AddDelta(parts, "HP", after.hp - before.hp); AddDelta(parts, "所持金", after.money - before.money); AddDelta(parts, "LUK", after.luck - before.luck); AddDelta(parts, "SAN", after.sanity - before.sanity); AddDelta(parts, "Mythos", after.mythosKnowledge - before.mythosKnowledge); AddDelta(parts, "Corruption", after.mythosCorruption - before.mythosCorruption); return string.Join(" / ", parts.ToArray()); }
        void AddDelta(List<string> parts, string label, int delta) { if (delta != 0) parts.Add(label + Signed(delta)); }
        string Signed(int value) { return value >= 0 ? "+" + value : value.ToString(); }

        string GearComparison(Gear next, Gear current) { return "現在装備との差分\n" + GearDeltaLine(next, current) + "\n\nNew: " + GearSideSummary(next) + "\nCurrent: " + GearSideSummary(current) + "\n\n" + next.note; }
        string GearDeltaLine(Gear next, Gear current) { int ca = current != null ? current.attack : 0, cd = current != null ? current.defense : 0, cs = current != null ? current.speed : 0, cl = current != null ? current.luck : 0; return "ATK" + Signed(next.attack - ca) + " / DEF" + Signed(next.defense - cd) + " / SPD" + Signed(next.speed - cs) + " / LUK" + Signed(next.luck - cl); }
